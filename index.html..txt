        // STATE APP
        let db = JSON.parse(localStorage.getItem('barakah_db')) || { time: {}, dzikir: {}, sunnah: {}, points: 0 };

        function saveDB() {
            localStorage.setItem('barakah_db', JSON.stringify(db));
            document.getElementById('total-points').innerText = db.points;
        }

        function switchView(viewId) {
            ['view-home', 'view-time', 'view-dzikir', 'view-sunnah100'].forEach(id => {
                document.getElementById(id).classList.add('hidden');
            });
            document.getElementById(viewId).classList.remove('hidden');
            window.scrollTo(0, 0);
        }

        // RENDER AMALAN WAKTU
        function renderTime() {
            const container = document.getElementById('time-list');
            container.innerHTML = timeData.map(item => {
                const checked = db.time[item.id] ? 'checked' : '';
                return `
                    <div class="glass-card p-4 rounded-xl space-y-2 border border-slate-700/50">
                        <div class="flex items-center justify-between">
                            <span class="bg-amber-500/10 text-amber-400 text-xs px-2 py-0.5 rounded font-bold">${item.time}</span>
                            <input type="checkbox" onchange="toggleTime('${item.id}')" ${checked} class="w-5 h-5 accent-amber-500 rounded cursor-pointer">
                        </div>
                        <h4 class="font-bold text-slate-100 text-sm">${item.title}</h4>
                        <div class="bg-slate-900/60 p-2.5 rounded-lg text-xs space-y-1">
                            <p class="text-amber-300 font-medium"><i class="fa-solid fa-gift mr-1"></i> ${item.reward}</p>
                            <p class="text-slate-400 text-[10px] italic">${item.dalil}</p>
                        </div>
                    </div>
                `;
            }).join('');
        }

        function toggleTime(id) {
            db.time[id] = !db.time[id];
            db.points += db.time[id] ? 10 : -10;
            saveDB();
        }

        // RENDER DZIKIR
        function renderDzikir() {
            const container = document.getElementById('dzikir-list');
            container.innerHTML = dzikirData.map(item => {
                const count = db.dzikir[item.id] || 0;
                return `
                    <div class="glass-card p-4 rounded-xl space-y-3 border border-slate-700/50">
                        <div class="flex justify-between items-start">
                            <h4 class="font-bold text-slate-100 text-sm">${item.title}</h4>
                            <span class="text-[10px] bg-slate-800 text-slate-400 px-2 py-1 rounded">${item.dalil}</span>
                        </div>
                        <p class="font-arabic text-right text-lg text-amber-300 leading-loose">${item.arab}</p>
                        <p class="text-xs text-slate-300 italic">${item.latin}</p>
                        <div class="flex items-center justify-between pt-2">
                            <span class="text-xs font-bold text-emerald-400">${count} / ${item.max}x</span>
                            <button onclick="addDzikir('${item.id}', ${item.max})" class="emerald-gradient text-slate-950 font-bold px-4 py-2 rounded-lg text-xs shadow hover:opacity-90 active:scale-95 transition-transform">
                                <i class="fa-solid fa-plus mr-1"></i> Hitung
                            </button>
                        </div>
                    </div>
                `;
            }).join('');
        }

        function addDzikir(id, max) {
            if (!db.dzikir[id]) db.dzikir[id] = 0;
            if (db.dzikir[id] < max) {
                db.dzikir[id]++;
                db.points += 1;
                saveDB();
                renderDzikir();
            }
        }

                        <input type="checkbox" onchange="toggleSunnah(${idx})" ${checked} class="mt-0.5 w-4 h-4 accent-purple-500 rounded">
                        <span class="text-xs text-slate-200 leading-relaxed"><strong class="text-purple-400 mr-1">${idx+1}.</strong> ${text}</span>
                    </label>
                `;
            }).join('');
        }

        function toggleSunnah(idx) {
            db.sunnah[idx] = !db.sunnah[idx];
            db.points += db.sunnah[idx] ? 5 : -5;
            saveDB();
        }

        function filterSunnah() {
            const q = document.getElementById('search-sunnah').value.toLowerCase();
            const items = document.querySelectorAll('#sunnah100-list label');
            items.forEach(item => {
                item.style.display = item.innerText.toLowerCase().includes(q) ? 'flex' : 'none';
            });
        }

        // INIT APP
        document.getElementById('total-points').innerText = db.points;
        renderTime();
        renderDzikir();
        renderSunnah();
    </script>
